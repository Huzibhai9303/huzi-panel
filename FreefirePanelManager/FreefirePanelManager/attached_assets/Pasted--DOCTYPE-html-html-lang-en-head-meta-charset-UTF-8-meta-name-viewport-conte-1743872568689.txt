      <p id="sensitivity-value" class="text-sm text-gray-400 mt-1">50%</p>

  <div id="crosshair"></div>

  <script>
    const sensitivityInput = document.getElementById('sensitivity');
    const sensitivityValue = document.getElementById('sensitivity-value');
    const output = document.getElementById('output');
    const bot = document.getElementById('bot');
    const crosshair = document.getElementById('crosshair');

    let botPosition = { x: 500, y: 300, headRadius: 25 }; // Bot's position

    sensitivityInput.addEventListener('input', () => {
      sensitivityValue.textContent = sensitivityInput.value + '%';
    });

    function aimbotEffect() {
      const aimbot = document.getElementById('aimbot').checked;
      const headshot = document.getElementById('headshot').checked;

      // Simulate aimbot effect
      if (aimbot) {
        crosshair.style.left = `${botPosition.x}px`;
        crosshair.style.top = `${botPosition.y}px`;

        if (headshot) {
          // Move to bot's head for headshot
          crosshair.style.top = `${botPosition.y - botPosition.headRadius}px`;
        }
      }
    }

    document.getElementById('startBtn').addEventListener('click', () => {
      const aimbot = document.getElementById('aimbot').checked;
      const headshot = document.getElementById('headshot').checked;
      const sensitivity = sensitivityInput.value;

      aimbotEffect();

      const result = `✔️ Script Running With:\n- Aimbot: ${aimbot ? 'Enabled' : 'Disabled'}\n- Auto Headshot: ${headshot ? 'Enabled' : 'Disabled'}\n- Sensitivity: ${sensitivity}%`;

      output.textContent = result;
    });
  </script>
